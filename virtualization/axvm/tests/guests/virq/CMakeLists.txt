cmake_minimum_required(VERSION 3.20.0)

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(software_virq_latency_suspend)

if(NOT DEFINED VIRQ_SAMPLES)
  set(VIRQ_SAMPLES 300)
endif()
if(NOT VIRQ_SAMPLES MATCHES "^[0-9]+$" OR VIRQ_SAMPLES LESS 1 OR VIRQ_SAMPLES GREATER 1000)
  message(FATAL_ERROR "VIRQ_SAMPLES must be between 1 and 1000")
endif()
target_compile_definitions(app PRIVATE SAMPLE_COUNT=${VIRQ_SAMPLES})
if(NOT DEFINED VIRQ_TASK)
  set(VIRQ_TASK 0)
endif()
if(NOT VIRQ_TASK MATCHES "^[01]$")
  message(FATAL_ERROR "VIRQ_TASK must be 0 or 1")
endif()
target_compile_definitions(app PRIVATE TASK_MODE=${VIRQ_TASK})
if(NOT DEFINED VIRQ_TASK_BLOCKING)
  set(VIRQ_TASK_BLOCKING 0)
endif()
if(NOT VIRQ_TASK_BLOCKING MATCHES "^[01]$")
  message(FATAL_ERROR "VIRQ_TASK_BLOCKING must be 0 or 1")
endif()
target_compile_definitions(app PRIVATE TASK_BLOCKING=${VIRQ_TASK_BLOCKING})
if(NOT DEFINED VIRQ_TASK_PERIODIC)
  set(VIRQ_TASK_PERIODIC 0)
endif()
if(NOT DEFINED VIRQ_TASK_ROUNDS)
  set(VIRQ_TASK_ROUNDS 4096)
endif()
if(NOT VIRQ_TASK_PERIODIC MATCHES "^[01]$" OR NOT VIRQ_TASK_ROUNDS MATCHES "^[0-9]+$" OR VIRQ_TASK_ROUNDS LESS 4096 OR VIRQ_TASK_ROUNDS GREATER 1048576)
  message(FATAL_ERROR "invalid periodic workload")
endif()
target_compile_definitions(app PRIVATE TASK_PERIODIC=${VIRQ_TASK_PERIODIC} TASK_ROUNDS=${VIRQ_TASK_ROUNDS})
target_sources(app PRIVATE src/main.c)
